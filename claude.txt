#### 3.2 Decomposition Strategy (Dynamic Programming Approach)
```python
class HierarchicalDecomposer:
    """Recursive module decomposition algorithm"""
    
    def __init__(self, max_tokens_per_module: int = 32768):
        self.max_tokens = max_tokens_per_module
        self.token_threshold = 0.8 * max_tokens  # 80% threshold
    
    def decompose(
        self, 
        components: Set[str],
        dependency_graph: EnhancedDependencyGraph,
        parent_name: str = "Repository"
    ) -> ModuleNode:
        """
        Recursively partition components into manageable modules
        
        Algorithm:
        1. Estimate token count for all components
        2. If under threshold, create leaf module
        3. Otherwise, partition and recurse
        """
        
        estimated_tokens = self._estimate_tokens(components)
        
        if estimated_tokens < self.token_threshold:
            # Base case: Create leaf module
            return ModuleNode(
                id=f"module_{uuid.uuid4().hex[:8]}",
                name=parent_name,
                components=components,
                children=[],
                parent=None,
                level=0,
                is_leaf=True,
                estimated_tokens=estimated_tokens,
                complexity_score=self._compute_complexity(components)
            )
        
        # Recursive case: Partition and decompose
        partitions = self._partition_components(components, dependency_graph)
        
        parent = ModuleNode(
            id=f"module_{uuid.uuid4().hex[:8]}",
            name=parent_name,
            components=set(),
            children=[],
            parent=None,
            level=0,
            is_leaf=False,
            estimated_tokens=estimated_tokens,
            complexity_score=0.0
        )
        
        for partition_name, partition_components in partitions.items():
            child = self.decompose(
                partition_components, 
                dependency_graph,
                partition_name
            )
            child.parent = parent
            parent.children.append(child)
        
        return parent
    
    def _partition_components(
```

#### 3.3 Partitioning Strategies
```python
class PartitioningStrategy:
    """Different approaches to module partitioning"""
    
    @staticmethod
    def by_directory_structure(components: Set[str]) -> Dict[str, Set[str]]:
        """Group by filesystem hierarchy"""
        partitions = defaultdict(set)
        for comp_id in components:
            # Extract directory from component path
            directory = Path(comp_id).parent.name
            partitions[directory].add(comp_id)
        return dict(partitions)
    
    @staticmethod
    def by_strongly_connected_components(
        components: Set[str],
        graph: nx.DiGraph
    ) -> Dict[str, Set[str]]:
        """Group tightly coupled components"""
        subgraph = graph.subgraph(components)
        sccs = nx.strongly_connected_components(subgraph)
        
        partitions = {}
        for idx, scc in enumerate(sccs):
            partitions[f"component_group_{idx}"] = scc
        return partitions
    
    @staticmethod
    def by_semantic_similarity(
        components: Set[str],
        embeddings: Dict[str, np.ndarray]
    ) -> Dict[str, Set[str]]:
        """Cluster by semantic similarity using embeddings"""
        from sklearn.cluster import AgglomerativeClustering
        
        comp_list = list(components)
        embedding_matrix = np.array([embeddings[c] for c in comp_list])
        
        clustering = AgglomerativeClustering(n_clusters=None, distance_threshold=0.5)
        labels = clustering.fit_predict(embedding_matrix)
        
        partitions = defaultdict(set)
        for comp, label in zip(comp_list, labels):
            partitions[f"cluster_{label}"].add(comp)
        
        return dict(partitions)
    
    @staticmethod
    def balanced_partition(
        components: Set[str],
        target_size: int,
        component_sizes: Dict[str, int]
    ) -> Dict[str, Set[str]]:
        """Partition to balance module sizes (bin packing)"""
        # Use First-Fit-Decreasing algorithm
        sorted_components = sorted(
            components, 
            key=lambda c: component_sizes[c],
            reverse=True
        )
        
        bins = []
        bin_sizes = []
        
        for comp in sorted_components:
            comp_size = component_sizes[comp]
            
            # Find bin with enough space
            placed = False
            for i, bin_size in enumerate(bin_sizes):
                if bin_size + comp_size <= target_size:
                    bins[i].add(comp)
                    bin_sizes[i] += comp_size
                    placed = True
                    break
            
            if not placed:
                bins.append({comp})
                bin_sizes.append(comp_size)
        
        return {f"partition_{i}": bin_set for i, bin_set in enumerate(bins)}
```

```python
class MultiModelRubricSynthesizer:

        1. Semantic similarity for requirement matching
        2. Weighted averaging for importance
        3. Structure alignment
        """

        self,
        rubrics: List[Dict[str, Any]]
    ) -> List[Dict[str, Any]]:
        """Flatten all rubrics to leaf requirements"""

            requirements = self._get_leaf_nodes(rubric)

        self,
        requirements: List[Dict[str, Any]],
        similarity_threshold: float = 0.85
    ) -> List[Dict[str, Any]]:
        """Remove duplicate requirements using embeddings"""
        from sentence_transformers import SentenceTransformer
        
        model = SentenceTransformer('all-MiniLM-L6-v2')
        
        # Embed all requirement descriptions
        texts = [req['description'] for req in requirements]
        embeddings = model.encode(texts)
        
        # Cluster similar requirements
        from sklearn.cluster import DBSCAN
        clustering = DBSCAN(eps=1-similarity_threshold, metric='cosine')
        labels = clustering.fit_predict(embeddings)
        
        # Keep one from each cluster
        unique_requirements = []
        seen_clusters = set()
        
        for req, label in zip(requirements, labels):
            if label not in seen_clusters:
                unique_requirements.append(req)
                seen_clusters.add(label)
        
        return unique_requirements
```

### Phase 6: Hierarchical Score Aggregation (Week 10)

#### 6.1 Score Propagation Algorithm
```python
class HierarchicalScoreAggregator:
    """Compute final scores with uncertainty propagation"""
    
    def aggregate_scores(
        self,
        rubric: Dict[str, Any],
        leaf_assessments: Dict[str, Dict[str, float]]
    ) -> Dict[str, Any]:
        """
        Bottom-up weighted aggregation with std dev propagation
        
        For leaf node ℓ with assessments s₁, s₂, ..., sₘ:
        S(ℓ) = mean(s₁, s₂, ..., sₘ)
        σ_ℓ = std(s₁, s₂, ..., sₘ)
        
        For internal node n with children c₁, ..., cₖ:
        S(n) = Σ(w(cᵢ) · S(cᵢ)) / Σ(w(cᵢ))
        σ_n = sqrt(Σ(w(cᵢ)² · σ²_cᵢ)) / Σ(w(cᵢ))
        """
        
        def aggregate_node(node: Dict[str, Any]) -> Tuple[float, float]:
            """Returns (score, std_dev) for a node"""
            
            if node.get('is_leaf', False):
                # Leaf node: get from assessments
                assessment = leaf_assessments[node['id']]
                return assessment['mean_score'], assessment['std_dev']
            
            # Internal node: aggregate from children
            child_scores = []
            child_stds = []
            child_weights = []
            
            for child in node['children']:
                score, std = aggregate_node(child)
                child_scores.append(score)
                child_stds.append(std)
                child_weights.append(child['weight'])
            
            # Weighted average for score
            total_weight = sum(child_weights)
            weighted_score = sum(
                w * s for w, s in zip(child_weights, child_scores)
            ) / total_weight
            
            # Propagate uncertainty
            weighted_variance = sum(
                (w ** 2) * (std ** 2) 
                for w, std in zip(child_weights, child_stds)
            ) / (total_weight ** 2)
            
            propagated_std = np.sqrt(weighted_variance)
            
            # Store in node
            node['score'] = weighted_score
            node['std_dev'] = propagated_std
            
            return weighted_score, propagated_std
        
        final_score, final_std = aggregate_node(rubric)
        
        return {
            "rubric": rubric,  # Now includes scores at all levels
            "final_score": final_score,
            "final_std_dev": final_std,
            "confidence": 1 - final_std  # Simple confidence measure
        }
```

#### 6.2 Reliability Metrics
```python
class ReliabilityAnalyzer:
    """Analyze reliability of evaluation methodology"""
    
    def compute_semantic_reliability(
        self,
        rubrics: List[Dict[str, Any]]
    ) -> float:
        """
        Compute semantic consistency across model-generated rubrics
        
        For rubrics R₁, R₂, ..., Rₙ:
        1. Extract all requirement texts
        2. Compute pairwise similarities
        3. Average across all pairs
        """
        from sentence_transformers import SentenceTransformer
        from sklearn.metrics.pairwise import cosine_similarity
        
        model = SentenceTransformer('all-MiniLM-L6-v2')
        
        # Extract requirements from each rubric
        all_requirements = []
        for rubric in rubrics:
            reqs = self._extract_leaf_requirements(rubric)
            all_requirements.append(reqs)
        
        # Compute pairwise similarities
        similarities = []
        n = len(rubrics)
        
        for i in range(n):
            for j in range(i + 1, n):
                sim = self._bidirectional_similarity(
                    all_requirements[i],
                    all_requirements[j],
                    model
                )
                similarities.append(sim)
        
        return np.mean(similarities)
    
    def _bidirectional_similarity(
        self,
        reqs_i: List[str],
        reqs_j: List[str],
        model
    ) -> float:
        """
        Compute bidirectional semantic similarity:
        sim(Rᵢ, Rⱼ) = 1/(mᵢ + mⱼ) * [Σ max_sim(tᵢ, Tⱼ) + Σ max_sim(tⱼ, Tᵢ)]
        """
        
        embeddings_i = model.encode(reqs_i)
        embeddings_j = model.encode(reqs_j)
        
        # For each req in i, find best match in j
        sims_i_to_j = []
        for emb_i in embeddings_i:
            sims = cosine_similarity([emb_i], embeddings_j)[0]
            sims_i_to_j.append(np.max(sims))
        
        # For each req in j, find best match in i
        sims_j_to_i = []
        for emb_j in embeddings_j:
            sims = cosine_similarity([emb_j], embeddings_i)[0]
            sims_j_to_i.append(np.max(sims))
        
        total = sum(sims_i_to_j) + sum(sims_j_to_i)
        return total / (len(reqs_i) + len(reqs_j))
    
    def compute_structural_reliability(
        self,
        rubrics: List[Dict[str, Any]]
    ) -> float:
        """
        Compute structural consistency:
        - Depth similarity
        - Item count similarity  
        - Weight distribution similarity
        """
        
        features = []
        for rubric in rubrics:
            features.append({
                'depth': self._max_depth(rubric),
                'items': self._count_items(rubric),
                'weights': self._weight_distribution(rubric)
            })
        
        # Compute pairwise structural similarities
        similarities = []
        n = len(rubrics)
        
        for i in range(n):
            for j in range(i + 1, n):
                depth_sim = 1 - abs(features[i]['depth'] - features[j]['depth']) / max(features[i]['depth'], features[j]['depth'])
                items_sim = 1 - abs(features[i]['items'] - features[j]['items']) / max(features[i]['items'], features[j]['items'])
                weights_sim = self._weight_overlap(features[i]['weights'], features[j]['weights'])
                
                avg_sim = (depth_sim + items_sim + weights_sim) / 3
                similarities.append(avg_sim)
        
        return np.mean(similarities)
```

```python
class CodeWikiBenchPipeline:
    """Complete evaluation pipeline"""
    
    def __init__(self, config: Dict[str, Any]):
        self.rubric_generators = config['rubric_models']
        self.judge_models = config['judge_models']
        self.parser = DocumentationParser()
        self.synthesizer = MultiModelRubricSynthesizer(self.rubric_generators)
        self.evaluator = MultiJudgeEvaluator(self.judge_models)
        self.aggregator = HierarchicalScoreAggregator()
    
    def evaluate_repository(
        self,
        repo_name: str,
        official_docs_path: Path,
        generated_docs_path: Path
    ) -> Dict[str, Any]:
        """
        Complete evaluation workflow:
        1. Parse official documentation
        2. Generate consensus rubric
        3. Evaluate generated docs
        4. Aggregate scores
        """
        
        print(f"Evaluating {repo_name}...")
        
        # Step 1: Parse official docs
        print("Parsing official documentation...")
        doc_structure = self.parser.parse_documentation_directory(official_docs_path)
        
        # Step 2: Generate rubric
        print("Generating evaluation rubric...")
        rubric = self.synthesizer.generate_consensus_rubric(
            doc_structure,
            {"name": repo_name}
        )
        
        # Step 3: Load generated documentation
        generated_docs = self._load_generated_docs(generated_docs_path)
        
        # Step 4: Evaluate each leaf requirement
        print("Evaluating requirements...")
        leaf_requirements = self._get_leaf_requirements(rubric)
        leaf_assessments = {}
        
        for req in tqdm(leaf_requirements):
            assessment = self.evaluator.evaluate_requirement(req, generated_docs)
            leaf_assessments[req['id']] = assessment
        
        # Step 5: Aggregate scores
        print("Aggregating scores...")
        results = self.aggregator.aggregate_scores(rubric, leaf_assessments)
        
        # Step 6: Generate report
        report = self._generate_report(repo_name, results)
                                       ```
